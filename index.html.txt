<section class="recipe">
  <div class="ingredients">
   <h2>
  <img src="https://cdn-icons-png.flaticon.com/512/7066/7066760.png" class="ingredients-icon">
  Ingredients
</h2>
    <ul class="list-ingredients"> 
    <li>3 eggs</li>
    <li>2 tomatoes</li>
    <li>2 green chili peppers</li>
    <li>1 tablespoon of butter</li>
    <li>Salt, black pepper, red pepper flakes</li>
  </ul>
  </div>
  <div class="preparation-box">
    <h2>
      <img src="https://www.vhv.rs/dpng/d/482-4820080_circle-hd-png-download.png" class="Preparation-icon">
      Preparation</h2>
  <ol class="preparation">
    <li>Melt the butter in a pan.</li>
    <li>Add the chopped peppers and sauté.</li>
    <li>Add the grated tomatoes and cook until the liquid reduces.</li>
    <li>Crack the eggs and stir.</li>
    <li>Add salt and spices, then serve.</li>
  </ol>
    </div>
